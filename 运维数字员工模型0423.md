## 一、用户与权限管理 (6张表)

-- ============================================================
-- 1. 用户表
-- ============================================================
CREATE TABLE `sys_user` (
  `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '用户ID',
  `username` VARCHAR(50) NOT NULL COMMENT '用户名',
  `password` VARCHAR(200) NOT NULL COMMENT '密码(加密)',
  `real_name` VARCHAR(50) NOT NULL COMMENT '真实姓名',
  `email` VARCHAR(100) NOT NULL COMMENT '邮箱',
  `phone` VARCHAR(20) DEFAULT NULL COMMENT '手机号',
  `avatar` VARCHAR(200) DEFAULT NULL COMMENT '头像URL',
  `department_id` BIGINT DEFAULT NULL COMMENT '部门ID',
  `job_title` VARCHAR(50) DEFAULT NULL COMMENT '职位',
  `employee_id` VARCHAR(50) DEFAULT NULL COMMENT '工号',
  `status` TINYINT NOT NULL DEFAULT 1 COMMENT '状态: 0-禁用 1-正常 2-锁定',
  `user_type` TINYINT NOT NULL DEFAULT 1 COMMENT '用户类型: 1-普通用户 2-管理员 3-超级管理员',
  `last_login_time` DATETIME DEFAULT NULL COMMENT '最后登录时间',
  `last_login_ip` VARCHAR(50) DEFAULT NULL COMMENT '最后登录IP',
  `login_count` INT DEFAULT 0 COMMENT '登录次数',
  `password_update_time` DATETIME DEFAULT NULL COMMENT '密码更新时间',
  `remark` VARCHAR(500) DEFAULT NULL COMMENT '备注',
  `create_by` VARCHAR(50) DEFAULT NULL COMMENT '创建人',
  `create_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `update_by` VARCHAR(50) DEFAULT NULL COMMENT '更新人',
  `update_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  `is_deleted` TINYINT NOT NULL DEFAULT 0 COMMENT '删除标记: 0-未删除 1-已删除',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_username` (`username`),
  UNIQUE KEY `uk_email` (`email`),
  KEY `idx_department` (`department_id`),
  KEY `idx_status` (`status`),
  KEY `idx_create_time` (`create_time`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='用户表';

-- ============================================================
-- 2. 角色表
-- ============================================================
CREATE TABLE `sys_role` (
  `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '角色ID',
  `role_code` VARCHAR(50) NOT NULL COMMENT '角色编码',
  `role_name` VARCHAR(50) NOT NULL COMMENT '角色名称',
  `role_level` INT NOT NULL DEFAULT 0 COMMENT '角色级别: 数字越小权限越大',
  `data_scope` TINYINT NOT NULL DEFAULT 1 COMMENT '数据范围: 1-全部 2-部门 3-部门及下级 4-本人',
  `status` TINYINT NOT NULL DEFAULT 1 COMMENT '状态: 0-禁用 1-正常',
  `description` VARCHAR(500) DEFAULT NULL COMMENT '角色描述',
  `remark` VARCHAR(500) DEFAULT NULL COMMENT '备注',
  `create_by` VARCHAR(50) DEFAULT NULL COMMENT '创建人',
  `create_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `update_by` VARCHAR(50) DEFAULT NULL COMMENT '更新人',
  `update_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  `is_deleted` TINYINT NOT NULL DEFAULT 0 COMMENT '删除标记',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_role_code` (`role_code`),
  KEY `idx_status` (`status`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='角色表';

-- ============================================================
-- 3. 权限表

## 二、任务管理 (8张表)
-- ============================================================

-- ============================================================
-- 7. 任务定义表
-- ============================================================
CREATE TABLE `task_definition` (
  `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '任务ID',
  `task_code` VARCHAR(100) NOT NULL COMMENT '任务编码',
  `task_name` VARCHAR(200) NOT NULL COMMENT '任务名称',
  `task_type` VARCHAR(50) NOT NULL COMMENT '任务类型: SQL/SHELL/PYTHON/SPARK等',
  `task_category` VARCHAR(50) DEFAULT NULL COMMENT '任务分类: ETL/报表/清洗等',
  `description` TEXT COMMENT '任务描述',
  `owner_id` BIGINT NOT NULL COMMENT '责任人ID',
  `department_id` BIGINT DEFAULT NULL COMMENT '所属部门',
  `priority` TINYINT NOT NULL DEFAULT 3 COMMENT '优先级: 1-最高 2-高 3-中 4-低 5-最低',
  `timeout` INT DEFAULT 3600 COMMENT '超时时间(秒)',
  `retry_times` INT DEFAULT 0 COMMENT '重试次数',
  `retry_interval` INT DEFAULT 60 COMMENT '重试间隔(秒)',
  `task_content` LONGTEXT COMMENT '任务内容/脚本',
  `task_params` TEXT COMMENT '任务参数(JSON)',
  `resource_config` TEXT COMMENT '资源配置(JSON)',
  `alert_config` TEXT COMMENT '告警配置(JSON)',
  `tags` VARCHAR(500) DEFAULT NULL COMMENT '标签(逗号分隔)',
  `status` TINYINT NOT NULL DEFAULT 1 COMMENT '状态: 0-禁用 1-启用 2-草稿',
  `version` INT NOT NULL DEFAULT 1 COMMENT '版本号',
  `is_published` TINYINT NOT NULL DEFAULT 0 COMMENT '是否发布: 0-未发布 1-已发布',
  `create_by` VARCHAR(50) DEFAULT NULL COMMENT '创建人',
  `create_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `update_by` VARCHAR(50) DEFAULT NULL COMMENT '更新人',
  `update_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  `is_deleted` TINYINT NOT NULL DEFAULT 0 COMMENT '删除标记',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_task_code` (`task_code`),
  KEY `idx_owner` (`owner_id`),
  KEY `idx_department` (`department_id`),
  KEY `idx_status` (`status`),
  KEY `idx_task_type` (`task_type`),
  KEY `idx_create_time` (`create_time`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='任务定义表';

-- ============================================================
-- 8. 任务实例表
-- ============================================================
CREATE TABLE `task_instance` (
  `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '实例ID',
  `task_id` BIGINT NOT NULL COMMENT '任务定义ID',
  `instance_code` VARCHAR(100) NOT NULL COMMENT '实例编码',
  `trigger_type` VARCHAR(50) NOT NULL COMMENT '触发类型: MANUAL/SCHEDULE/DEPEND/API',
  `trigger_time` DATETIME NOT NULL COMMENT '触发时间',
  `start_time` DATETIME DEFAULT NULL COMMENT '开始时间',
  `end_time` DATETIME DEFAULT NULL COMMENT '结束时间',
  `duration` INT DEFAULT NULL COMMENT '执行时长(秒)',
  `status` VARCHAR(50) NOT NULL DEFAULT 'WAITING' COMMENT '状态: WAITING/RUNNING/SUCCESS/FAILED/KILLED/TIMEOUT',
  `execute_host` VARCHAR(100) DEFAULT NULL COMMENT '执行主机',
  `execute_user` VARCHAR(50) DEFAULT NULL COMMENT '执行用户',
  `process_id` VARCHAR(50) DEFAULT NULL COMMENT '进程ID',
  `log_path` VARCHAR(500) DEFAULT NULL COMMENT '日志路径',
  `error_message` TEXT COMMENT '错误信息',
  `retry_count` INT DEFAULT 0 COMMENT '重试次数',
  `progress` INT DEFAULT 0 COMMENT '执行进度(%)',
  `input_records` BIGINT DEFAULT 0 COMMENT '输入记录数',
  `output_records` BIGINT DEFAULT 0 COMMENT '输出记录数',
  `error_records` BIGINT DEFAULT 0 COMMENT '错误记录数',
  `task_params` TEXT COMMENT '任务参数(JSON)',
  `result_data` TEXT COMMENT '结果数据(JSON)',
  `create_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `update_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_instance_code` (`instance_code`),
  KEY `idx_task_id` (`task_id`),
  KEY `idx_status` (`status`),
  KEY `idx_trigger_time` (`trigger_time`),
  KEY `idx_create_time` (`create_time`),
  KEY `idx_start_time` (`start_time`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='任务实例表';

-- ============================================================
-- 9. 任务调度配置表

CREATE TABLE `task_schedule` (
  `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '调度ID',

  `schedule_type` VARCHAR(50) NOT NULL COMMENT '调度类型: CRON/FIXED_RATE/ONCE',

  `condition_expression` TEXT COMMENT '条件表达式',
  `check_interval` INT DEFAULT 300 COMMENT '检查间隔(秒)',
  `alert_interval` INT DEFAULT 1800 COMMENT '告警间隔(秒)',
  `max_alert_count` INT DEFAULT 10 COMMENT '最大告警次数',
  `recovery_notice` TINYINT DEFAULT 1 COMMENT '是否恢复通知: 0-否 1-是',
  `notification_config` TEXT COMMENT '通知配置(JSON)',
  `owner_id` BIGINT NOT NULL COMMENT '责任人ID',
  `status` TINYINT NOT NULL DEFAULT 1 COMMENT '状态: 0-禁用 1-启用',
  `remark` VARCHAR(500) DEFAULT NULL COMMENT '备注',
  `create_by` VARCHAR(50) DEFAULT NULL COMMENT '创建人',
  `create_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `update_by` VARCHAR(50) DEFAULT NULL COMMENT '更新人',
  `update_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  `is_deleted` TINYINT NOT NULL DEFAULT 0 COMMENT '删除标记',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_rule_code` (`rule_code`),
  KEY `idx_rule_type` (`rule_type`),
  KEY `idx_alert_level` (`alert_level`),

-- ============================================================
-- 16. 告警记录表
-- ============================================================
CREATE TABLE `alert_record` (
  `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '告警ID',
  `alert_code` VARCHAR(100) NOT NULL COMMENT '告警编码',
  `rule_id` BIGINT NOT NULL COMMENT '规则ID',
  `alert_title` VARCHAR(200) NOT NULL COMMENT '告警标题',
  `alert_content` TEXT NOT NULL COMMENT '告警内容',
  `alert_level` VARCHAR(20) NOT NULL COMMENT '告警级别',
  `alert_source` VARCHAR(100) DEFAULT NULL COMMENT '告警来源',
  `task_id` BIGINT DEFAULT NULL COMMENT '关联任务ID',
  `instance_id` BIGINT DEFAULT NULL COMMENT '关联实例ID',
  `metric_value` VARCHAR(200) DEFAULT NULL COMMENT '指标值',
  `threshold_value` VARCHAR(200) DEFAULT NULL COMMENT '阈值',
  `alert_time` DATETIME NOT NULL COMMENT '告警时间',
  `status` VARCHAR(50) NOT NULL DEFAULT 'PENDING' COMMENT '状态: PENDING/PROCESSING/CONFIRMED/IGNORED/RESOLVED',
  `handler_id` BIGINT DEFAULT NULL COMMENT '处理人ID',

## 十一、视图创建
-- ============================================================

-- 1. 任务执行统计视图
CREATE OR REPLACE VIEW v_task_execution_stats AS
SELECT 
    t.id AS task_id,
    t.task_code,
    t.task_name,
    t.task_type,
    COUNT(ti.id) AS total_count,
    SUM(CASE WHEN ti.status = 'SUCCESS' THEN 1 ELSE 0 END) AS success_count,
    SUM(CASE WHEN ti.status = 'FAILED' THEN 1 ELSE 0 END) AS failed_count,
    ROUND(SUM(CASE WHEN ti.status = 'SUCCESS' THEN 1 ELSE 0 END) / COUNT(ti.id) * 100, 2) AS success_rate,
    AVG(ti.duration) AS avg_duration,
    MAX(ti.duration) AS max_duration,
    MIN(ti.duration) AS min_duration
FROM task_definition t
LEFT JOIN task_instance ti ON t.id = ti.task_id
WHERE ti.create_time >= DATE_SUB(CURDATE(), INTERVAL 30 DAY)
GROUP BY t.id, t.task_code, t.task_name, t.task_type;

-- 2. 告警处理效率视图
CREATE OR REPLACE VIEW v_alert_handle_efficiency AS
SELECT 
    ar.handler_id,
    u.real_name AS handler_name,
    COUNT(ar.id) AS total_alerts,
    SUM(CASE WHEN ar.status = 'RESOLVED' THEN 1 ELSE 0 END) AS resolved_count,
    AVG(TIMESTAMPDIFF(SECOND, ar.alert_time, ar.handle_time)) AS avg_response_time,
    AVG(TIMESTAMPDIFF(SECOND, ar.alert_time, ar.resolve_time)) AS avg_resolve_time
FROM alert_record ar
LEFT JOIN sys_user u ON ar.handler_id = u.id
WHERE ar.alert_time >= DATE_SUB(CURDATE(), INTERVAL 30 DAY)
  AND ar.handler_id IS NOT NULL
GROUP BY ar.handler_id, u.real_name;

-- 3. 数据资产概览视图
CREATE OR REPLACE VIEW v_data_asset_overview AS
SELECT 
    mt.business_domain,
    COUNT(mt.id) AS table_count,
    SUM(mt.row_count) AS total_rows,
    SUM(mt.data_size) AS total_size,
    AVG(mt.quality_score) AS avg_quality_score,
    COUNT(DISTINCT mt.owner_id) AS owner_count
FROM metadata_table mt
WHERE mt.is_deleted = 0 AND mt.is_active = 1
GROUP BY mt.business_domain;

-- 4. 用户活跃度视图
CREATE OR REPLACE VIEW v_user_activity AS
SELECT 
    u.id AS user_id,
    u.username,
    u.real_name,
    d.dept_name,
    COUNT(DISTINCT DATE(ol.operation_time)) AS active_days,
    COUNT(ol.id) AS operation_count,
    MAX(ol.operation_time) AS last_operation_time
FROM sys_user u
LEFT JOIN sys_department d ON u.department_id = d.id
LEFT JOIN sys_operation_log ol ON u.id = ol.operator_id
WHERE ol.operation_time >= DATE_SUB(CURDATE(), INTERVAL 30 DAY)
GROUP BY u.id, u.username, u.real_name, d.dept_name;

DELIMITER $$
CREATE TRIGGER trg_task_instance_after_update
AFTER UPDATE ON task_instance
FOR EACH ROW
BEGIN
    -- 当任务状态变为失败时，自动创建告警
    IF NEW.status = 'FAILED' AND OLD.status != 'FAILED' THEN
        INSERT INTO alert_record (
            alert_code, 
            rule_id, 
            alert_title, 
            alert_content, 
            alert_level,
            task_id,
            instance_id,
            alert_time,
            status
        ) VALUES (

-- 1. 任务依赖检查存储过程
DELIMITER $$
CREATE PROCEDURE check_task_dependencies(
    IN p_task_id BIGINT,
    OUT p_can_execute BOOLEAN
)
BEGIN
    DECLARE done INT DEFAULT FALSE;
    DECLARE v_depend_task_id BIGINT;
    DECLARE v_depend_status VARCHAR(50);
  
    DECLARE cur CURSOR FOR 
        SELECT depend_task_id 
        FROM task_dependency 
        WHERE task_id = p_task_id AND status = 1;
  
    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = TRUE;
  
    SET p_can_execute = TRUE;
  
    OPEN cur;
    read_loop: LOOP
        FETCH cur INTO v_depend_task_id;
        IF done THEN
            LEAVE read_loop;
        END IF;
      
        -- 检查依赖任务的最新实例状态
        SELECT status INTO v_depend_status
        FROM task_instance
        WHERE task_id = v_depend_task_id
        ORDER BY create_time DESC
        LIMIT 1;
      
        -- 如果依赖任务未成功，则不能执行
        IF v_depend_status IS NULL OR v_depend_status != 'SUCCESS' THEN
            SET p_can_execute = FALSE;
            LEAVE read_loop;
        END IF;
    END LOOP;
  
    CLOSE cur;
END$$
DELIMITER ;

-- 2. 告警统计聚合存储过程
DELIMITER $$
CREATE PROCEDURE aggregate_alert_statistics(IN p_date DATE)
BEGIN
    -- 按小时统计
    INSERT INTO alert_statistics (
        stat_date,
        stat_hour,
        alert_level,
        alert_type,
        total_count,
        pending_count,
        processing_count,
        resolved_count,
        ignored_count,
        avg_handle_time
    )
    SELECT 
        DATE(alert_time) AS stat_date,
        HOUR(alert_time) AS stat_hour,
        alert_level,
        'ALL' AS alert_type,
        COUNT(*) AS total_count,
        SUM(CASE WHEN status = 'PENDING' THEN 1 ELSE 0 END) AS pending_count,
        SUM(CASE WHEN status = 'PROCESSING' THEN 1 ELSE 0 END) AS processing_count,
        SUM(CASE WHEN status = 'RESOLVED' THEN 1 ELSE 0 END) AS resolved_count,
        SUM(CASE WHEN status = 'IGNORED' THEN 1 ELSE 0 END) AS ignored_count,
        AVG(TIMESTAMPDIFF(SECOND, alert_time, handle_time)) AS avg_handle_time
    FROM alert_record
    WHERE DATE(alert_time) = p_date
    GROUP BY DATE(alert_time), HOUR(alert_time), alert_level
    ON DUPLICATE KEY UPDATE
        total_count = VALUES(total_count),
        pending_count = VALUES(pending_count),
        processing_count = VALUES(processing_count),
        resolved_count = VALUES(resolved_count),
        ignored_count = VALUES(ignored_count),
        avg_handle_time = VALUES(avg_handle_time);
END$$
DELIMITER ;

-- 3. 慢查询分析
CREATE OR REPLACE VIEW v_slow_query_analysis AS
SELECT 
    DIGEST_TEXT AS query_pattern,
    COUNT_STAR AS exec_count,
    ROUND(AVG_TIMER_WAIT / 1000000000000, 2) AS avg_time_sec,
    ROUND(MAX_TIMER_WAIT / 1000000000000, 2) AS max_time_sec,
    ROUND(SUM_ROWS_EXAMINED / COUNT_STAR, 0) AS avg_rows_examined,
    ROUND(SUM_ROWS_SENT / COUNT_STAR, 0) AS avg_rows_sent,
    FIRST_SEEN,
    LAST_SEEN
FROM performance_schema.events_statements_summary_by_digest
WHERE SCHEMA_NAME = 'dataops_db'
  AND AVG_TIMER_WAIT > 1000000000000  -- >1秒
ORDER BY AVG_TIMER_WAIT DESC
LIMIT 20;

-- 4. 索引使用分析
SELECT 
    t.TABLE_NAME AS '表名',
    s.INDEX_NAME AS '索引名',
    s.CARDINALITY AS '基数',
    ROUND(s.CARDINALITY / t.TABLE_ROWS * 100, 2) AS '选择性(%)'
FROM information_schema.STATISTICS s
JOIN information_schema.TABLES t 
    ON s.TABLE_SCHEMA = t.TABLE_SCHEMA 
    AND s.TABLE_NAME = t.TABLE_NAME
WHERE s.TABLE_SCHEMA = 'dataops_db'
  AND s.INDEX_NAME != 'PRIMARY'
ORDER BY s.TABLE_NAME, s.CARDINALITY DESC;
